---
title: Draggable Block
description: Drag handle that appears next to blocks for reordering
---

Adds a grab handle to the left of each block on hover. Drag to reorder
paragraphs, headings, lists, quotes, code blocks, anything block-level.

## Install

```bash
pnpm add @typix-editor/extension-draggable-block
```

## Live demo

<ComponentPreview name="draggable-block" />

## Custom class names

```tsx
import type { DraggableBlockClassNames } from "@typix-editor/extension-draggable-block";

const classNames: DraggableBlockClassNames = {
  handle: "my-handle",
  handleActive: "my-handle-active",
  targetLine: "my-drop-line",
};

<DraggableBlock classNames={classNames} />
```

## Vendor the UI

```bash
npx typix ui add draggable-block
```
